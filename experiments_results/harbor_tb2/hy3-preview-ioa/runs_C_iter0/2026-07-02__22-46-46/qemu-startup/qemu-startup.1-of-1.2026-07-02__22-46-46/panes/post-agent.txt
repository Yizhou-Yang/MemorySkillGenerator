import socket, threading, time, os
QHOST='127.0.0.1'; QPORT=4545; LPORT=6665
READY='/app/run/ready'
BUF=bytearray()
buf_lock=threading.Lock()
state={'client':None}
state_lock=threading.Lock()
qc=None
def escape_iac(d): return d.replace(b'\xff', b'\xff\xff')
def connect_qemu():
    while True:
        try:
            s=socket.create_connection((QHOST,QPORT),timeout=5); s.settimeout(0.5); return s
        except Exception:
            time.sleep(0.3)
def qemu_loop():
    global qc
    while True:
        if qc is None:
            try: qc=connect_qemu(); print('relay: connected to qemu serial', flush=True)
            except Exception: time.sleep(0.3); continue
        try: data=qc.recv(65536)
        except socket.timeout: continue
        except Exception: qc=None; continue
        if not data: qc=None; continue
        with buf_lock:
            BUF.extend(data); buf=bytes(BUF)
        with state_lock: c=state['client']
        if c is not None:
            try: c.sendall(escape_iac(data))
            except Exception:
                with state_lock: state['client']=None
        if not os.path.exists(READY) and b'login:' in buf.lower():
            try: open(READY,'w').write('ready\n')
            except Exception: pass
            print('relay: READY login prompt seen', flush=True)
def client_loop(conn):
    with state_lock: state['client']=conn
    with buf_lock: hist=bytes(BUF)
    try: conn.sendall(escape_iac(hist))
    except Exception: pass
    iac=False; iac_cmd=None; sub=False
    while True:
        try: chunk=conn.recv(4096)
        except Exception: break
        if not chunk: break
        out=bytearray()
        for b in chunk:
            if sub:
                if b==240: sub=False
                continue
            if iac:
                if iac_cmd is None:
                    iac_cmd=b
                    if b==250: sub=True; iac=False
                    elif b in (240,241,242,243,244): iac=False
                else:
                    opt=b
                    try:
                        if iac_cmd in (251,252): conn.sendall(bytes([255,253,opt]))
                        elif iac_cmd in (253,254): conn.sendall(bytes([255,251,opt]))
                    except Exception: pass
                    iac=False; iac_cmd=None
                continue
            if b==255: iac=True; iac_cmd=None; continue
            out.append(b)
        if out and qc is not None:
            try: qc.sendall(bytes(out))
            except Exception: pass
    with state_lock:
        if state['client'] is conn: state['client']=None
    try: conn.close()
    except Exception: pass
def main():
    threading.Thread(target=qemu_loop, daemon=True).start()
    ls=socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    ls.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR,1)
    ls.bind((QHOST,LPORT)); ls.listen(1)
    print('relay: listening on %d' % LPORT, flush=True)
    while True:
        conn,_=ls.accept()
        threading.Thread(target=client_loop, args=(conn,), daemon=True).start()
main()
