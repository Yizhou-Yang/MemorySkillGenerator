#!/usr/bin/env python3
import subprocess, sys
sys.setrecursionlimit(100000)

DELTA = {'N':(-1,0),'S':(1,0),'E':(0,1),'W':(0,-1)}
OPP = {'N':'S','S':'N','E':'W','W':'E'}
ORDER = ['N','S','E','W']

proc = subprocess.Popen(['python3','-u','/protected/maze_server.py'],
    stdin=subprocess.PIPE, stdout=subprocess.PIPE, text=True, bufsize=1)

def readline():
    return proc.stdout.readline().rstrip('\n')

def getres():
    line = readline()
    if '> ' in line:
        line = line.split('> ',1)[1]
    return line.strip()

for _ in range(4):
    readline()

visited=set()
walls=set()
celltype={}

def send(cmd):
    proc.stdin.write(cmd+'\n')
    proc.stdin.flush()

def explore(pos):
    if pos in visited:
        return
    visited.add(pos)
    if pos==(0,0):
        celltype[pos]='S'
    for d in ORDER:
        send('move '+d)
        r=getres()
        if r=='hit wall':
            neigh=(pos[0]+DELTA[d][0], pos[1]+DELTA[d][1])
            walls.add(neigh)
        elif r in ('moved','reached exit'):
            neigh=(pos[0]+DELTA[d][0], pos[1]+DELTA[d][1])
            if neigh not in celltype:
                celltype[neigh]= 'E' if r=='reached exit' else ' '
            explore(neigh)
            send('move '+OPP[d])
            getres()
        else:
            print('UNEXPECTED:', repr(r), file=sys.stderr)

explore((0,0))
