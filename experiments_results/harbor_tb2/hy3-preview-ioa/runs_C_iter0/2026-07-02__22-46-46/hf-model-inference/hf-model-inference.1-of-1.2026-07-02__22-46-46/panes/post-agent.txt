> import os
> from flask import Flask, request, jsonify
> from transformers import pipeline
>
> MODEL_DIR = "/app/model_cache/sentiment_model"
>
> app = Flask(__name__)
>
> print("Loading sentiment model from", MODEL_DIR, flush=True)
> classifier = pipeline(
>     "sentiment-analysis",
>     model=MODEL_DIR,
>     tokenizer=MODEL_DIR,
> )
> print("Model loaded.", flush=True)
>
>
> def build_response(label, score):
>     # distilbert-sst-2 labels are POSITIVE / NEGATIVE
>     pos = score if label.upper().startswith("POS") else 1.0 - score
>     neg = score if label.upper().startswith("NEG") else 1.0 - score
>     sentiment = "positive" if label.upper().startswith("POS") else "negative"
>     return {
>         "sentiment": sentiment,
>         "confidence": {
>             "positive": round(float(pos), 6),
>             "negative": round(float(neg), 6),
>         },
>     }
>
>
> @app.route("/sentiment", methods=["POST"])
> def sentiment():
>     try:
>         data = request.get_json(silent=True)
>         if not data or "text" not in data:
>             return jsonify({"error": "Missing 'text' field in JSON body"}), 400
>         text = data["text"]
>         if not isinstance(text, str) or text.strip() == "":
>             return jsonify({"error": "'text' must be a non-empty string"}), 400
>         results = classifier(text)
>         res = results[0]
>         return jsonify(build_response(res["label"], res["score"]))
>     except Exception as e:
>         return jsonify({"error": str(e)}), 500
>
>
> @app.route("/health", methods=["GET"])
> def health():
>     return jsonify({"status": "ok"})
>
>
> if __name__ == "__main__":
>     app.run(host="0.0.0.0", port=5000)
> EOF
