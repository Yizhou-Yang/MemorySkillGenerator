> #!/usr/bin/env python3
> import ssl
> import subprocess
> import os
> from datetime import datetime
>
> CERT_PATH = "/app/ssl/server.crt"
>
> def main():
>     if not os.path.exists(CERT_PATH):
>         print(f"Certificate not found at {CERT_PATH}")
>         return
>
>     with open(CERT_PATH) as f:
>         cert_data = f.read()
>
>     # Load/verify the certificate can be parsed via the ssl module
>     der = ssl.PEM_cert_to_DER_cert(cert_data)
>
>     result = subprocess.run(
>         ["openssl", "x509", "-in", CERT_PATH, "-noout", "-subject", "-enddate"],
>         capture_output=True, text=True,
>     )
>     if result.returncode != 0:
>         print("Failed to load certificate")
>         return
>
>     subject = ""
>     enddate = ""
>     for line in result.stdout.splitlines():
>         line = line.strip()
>         if line.startswith("subject="):
>             subject = line[len("subject="):].strip()
>         elif line.startswith("notAfter="):
>             enddate = line[len("notAfter="):].strip()
>
>     cn = "Unknown"
>     for part in subject.split(","):
>         part = part.strip()
>         if part.startswith("CN"):
>             cn = part.split("=", 1)[1].strip()
>
>     # Convert expiration date to YYYY-MM-DD (drop timezone token)
>     dt = datetime.strptime(enddate.rsplit(" ", 1)[0], "%b %d %H:%M:%S %Y")
>     expiration = dt.strftime("%Y-%m-%d")
>
>     print(f"Common Name: {cn}")
>     print(f"Expiration date: {expiration}")
>     print("Certificate verification successful")
>
> if __name__ == "__main__":
>     main()
> PYEOF
