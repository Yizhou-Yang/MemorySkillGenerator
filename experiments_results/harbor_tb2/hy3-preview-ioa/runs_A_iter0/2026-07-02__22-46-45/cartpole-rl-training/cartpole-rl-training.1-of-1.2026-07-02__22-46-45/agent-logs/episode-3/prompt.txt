import os
import torch
import torch.nn as nn


class DQN(nn.Module):
    def __init__(self, n_obs=4, n_actions=2):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(n_obs, 128),
            nn.ReLU(),
            nn.Linear(128, 64),
            nn.ReLU(),
            nn.Linear(64, n_actions),
        )

    def forward(self, x):
        return self.net(x)


class Agent:
    def __init__(self, n_obs=4, n_actions=2):
        self.n_obs = n_obs
        self.n_actions = n_actions
        self.device = torch.device("cpu")
        self.net = DQN(n_obs, n_actions).to(self.device)
        self.net.eval()

    def select_action(self, current_observation):
        obs = torch.as_tensor(current_observation, dtype=torch.float32).to(self.device)
        if obs.dim() == 1:
            obs = obs.unsqueeze(0)
        with torch.no_grad():
            q = self.net(obs)
            action = int(q.argmax(dim=1).item())
        return action

    def save(self, directory):
        os.makedirs(directory, exist_ok=True)
        path = os.path.join(directory, "model.pt")
        torch.save(self.net.state_dict(), path)

    def load(self, directory):
        path = os.path.join(directory, "model.pt")
        state = torch.load(path, map_location=self.device)
        self.net.load_state_dict(state)
        self.net.eval()
