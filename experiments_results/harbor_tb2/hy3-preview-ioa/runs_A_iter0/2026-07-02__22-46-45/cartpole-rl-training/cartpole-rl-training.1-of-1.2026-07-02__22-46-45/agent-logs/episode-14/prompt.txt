import os, random, math
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
from collections import deque
import gymnasium as gym
from agent import Agent, DQN

SEED = 0
random.seed(SEED); np.random.seed(SEED); torch.manual_seed(SEED)

gamma = 0.99
lr = 5e-4
batch_size = 128
buffer_size = 50000
learning_starts = 500
train_freq = 1
target_update = 300
eps_start = 1.0
eps_end = 0.02
eps_decay = 3000
max_episodes = 4000
max_steps = 500
best_dir = "/app/trained_model"

env = gym.make("CartPole-v1")
env.reset(seed=SEED)

agent = Agent()
optimizer = optim.Adam(agent.net.parameters(), lr=lr)
target_net = DQN().to(agent.device)
target_net.load_state_dict(agent.net.state_dict())
target_net.eval()

replay = deque(maxlen=buffer_size)
steps_done = 0

def select_epsilon_action(obs):
    global steps_done
    eps = eps_end + (eps_start - eps_end) * math.exp(-steps_done / eps_decay)
    steps_done += 1
    if random.random() < eps:
        return env.action_space.sample()
    return agent.select_action(obs)

def optimize():
    if len(replay) < batch_size:
        return
    batch = random.sample(replay, batch_size)
    obs, acts, rewards, next_obs, dones = zip(*batch)
    obs = torch.as_tensor(np.array(obs), dtype=torch.float32)
    acts = torch.as_tensor(acts, dtype=torch.int64).unsqueeze(1)
    rewards = torch.as_tensor(rewards, dtype=torch.float32).unsqueeze(1)
    next_obs = torch.as_tensor(np.array(next_obs), dtype=torch.float32)
    dones = torch.as_tensor(dones, dtype=torch.float32).unsqueeze(1)
    q_vals = agent.net(obs).gather(1, acts)
    with torch.no_grad():
        next_q = target_net(next_obs).max(1)[0].unsqueeze(1)
        target = rewards + gamma * next_q * (1 - dones)
    loss = nn.functional.smooth_l1_loss(q_vals, target)
    optimizer.zero_grad(); loss.backward(); optimizer.step()

def evaluate(n=30):
    rs = []
    for _ in range(n):
        obs, _ = env.reset()
        tot = 0
        for _ in range(max_steps):
            a = agent.select_action(obs)
            obs, r, term, trunc, _ = env.step(a)
            tot += r
            if term or trunc: break
        rs.append(tot)
    return sum(rs)/len(rs)

best_avg = -1
for ep in range(max_episodes):
    obs, _ = env.reset()
    for t in range(max_steps):
        a = select_epsilon_action(obs)
        no, r, term, trunc, _ = env.step(a)
        replay.append((obs, a, r, no, term or trunc))
        obs = no
        if steps_done > learning_starts and steps_done % train_freq == 0:
            optimize()
        if steps_done > learning_starts and steps_done % target_update == 0:
            target_net.load_state_dict(agent.net.state_dict())
        if term or trunc: break
    if steps_done > learning_starts and ep % 25 == 0:
        avg = evaluate(30)
        print(f"Episode {ep}, steps {steps_done}, eval_avg {avg:.1f}", flush=True)
        if avg > best_avg:
            best_avg = avg
            agent.save(best_dir)
            print(f"  -> saved best ({avg:.1f})", flush=True)
        if avg >= 450:
            print("Target reached, stopping.", flush=True)
            break

if best_avg < 0:
    agent.save(best_dir)
print(f"Best eval avg: {best_avg:.1f}", flush=True)
env.close()
