>

>     model=MODEL_PATH,
>     tokenizer=MODEL_PATH,
> )
> print("Model loaded.", flush=True)
>
>
> @app.route("/sentiment", methods=["POST"])
> def sentiment():
>     try:
>         data = request.get_json(silent=True)
>         if not data or not isinstance(data, dict):
>             return jsonify({"error": "Invalid JSON body"}), 400
>         text = data.get("text")
>         if text is None or not isinstance(text, str) or text.strip() == "":
>             return jsonify({"error": "Missing or empty 'text' field"}), 400
>
>         results = classifier(text)
>         if isinstance(results, list):
>             results = results[0]
>
>         label = results["label"].lower()
>         score = float(results["score"])
>
>         # Normalize labels: POSITIVE -> positive, NEGATIVE -> negative
>         is_positive = "positive" in label
>         confidence = {
>             "positive": score if is_positive else round(1.0 - score, 6),
>             "negative": round(1.0 - score, 6) if is_positive else score,
>         }
>         sentiment_label = "positive" if is_positive else "negative"
>
>         return jsonify({
>             "sentiment": sentiment_label,
>             "confidence": confidence,
>         })
>     except Exception as e:
>         traceback.print_exc()
>         return jsonify({"error": str(e)}), 500
>
>
> @app.route("/health", methods=["GET"])
> def health():
>     return jsonify({"status": "ok"})
>
>
