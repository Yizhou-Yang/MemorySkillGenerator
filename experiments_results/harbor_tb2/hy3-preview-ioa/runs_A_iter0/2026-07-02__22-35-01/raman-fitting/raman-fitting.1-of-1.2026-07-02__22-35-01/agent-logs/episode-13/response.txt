import numpy as np, json
from scipy.optimize import curve_fit

x=[]; y=[]
with open('/app/graphene.dat') as f:
    for line in f:
        p=line.replace(',','.').split()
        if len(p)>=2:
            x.append(float(p[0])); y.append(float(p[1]))
x=np.array(x); y=np.array(y)
idx=np.argsort(x); x=x[idx]; y=y[idx]

def lor(x,x0,gamma,amp,offset):
    return offset + amp*gamma**2/((x-x0)**2+gamma**2)

def fit_window(xc, dx, p0):
    m=(x>xc-dx)&(x<xc+dx)
    xs=x[m]; ys=y[m]
    popt,pcov=curve_fit(lor,xs,ys,p0=p0,
        bounds=([xc-3000,1,0,0],[xc+3000,2000,5e5,1e5]),maxfev=200000)
    yfit=lor(xs,*popt)
    ss=1-((ys-yfit)**2).sum()/((ys-ys.mean())**2).sum()
    return popt, ss

# 2D peak
p2d,r2d=fit_window(19139,2200,[19139,450,72000,15000])
# G peak
pG,rG=fit_window(10289,1800,[10289,350,10000,10000])
print('2D x0=%.2f gamma=%.2f amp=%.2f offset=%.2f  R2=%.5f'%(*p2d,r2d))
print('G  x0=%.2f gamma=%.2f amp=%.2f offset=%.2f  R2=%.5f'%(*pG,rG))

res={
 "G":{"x0":float(pG[0]),"gamma":float(pG[1]),"amplitude":float(pG[2]),"offset":float(pG[3])},
 "2D":{"x0":float(p2d[0]),"gamma":float(p2d[1]),"amplitude":float(p2d[2]),"offset":float(p2d[3])}
}
with open('/app/results.json','w') as f:
    json.dump(res,f,indent=2)
print(json.dumps(res,indent=2))
