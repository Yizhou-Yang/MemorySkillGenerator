> #!/usr/bin/env python3
> '''reshard.py
>
> Prepare c4_sample/ for upload:
>   * at most 30 files per folder
>   * at most 15MB per file
>
> Files are moved into c4_reshard/<bucket>/ (<=30 files per bucket). Any file
> larger than 15MB is split into <=15MB chunks. A manifest.json records how to
> reconstruct each original file so revert.py can restore c4_sample/ exactly.
> '''
> import json
> import os
> import shutil
>
> SAMPLE = '/app/c4_sample'
> RESHARD = '/app/c4_reshard'
> MAX_FILES = 30
> MAX_SIZE = 15 * 1024 * 1024
>
>
> def main():
>     if os.path.exists(RESHARD):
>         shutil.rmtree(RESHARD)
>     os.makedirs(RESHARD)
>
>     files = sorted(
>         name for name in os.listdir(SAMPLE)
>         if os.path.isfile(os.path.join(SAMPLE, name))
>     )
>
>     manifest = []
>     bucket_idx = 0
>     count = 0
>
>     def new_bucket():
>         nonlocal bucket_idx, count
>         bucket_idx += 1
>         count = 0
>         os.makedirs(os.path.join(RESHARD, '%03d' % bucket_idx), exist_ok=True)
>
>     os.makedirs(os.path.join(RESHARD, '000'), exist_ok=True)
>
>     for name in files:
>         src = os.path.join(SAMPLE, name)
>         size = os.path.getsize(src)
>         parts = []
>         if size <= MAX_SIZE:
>             if count >= MAX_FILES:
>                 new_bucket()
>             dst = os.path.join(RESHARD, '%03d' % bucket_idx, name)
>             shutil.move(src, dst)
>             parts.append('%03d/%s' % (bucket_idx, name))
>             count += 1
>         else:
>             with open(src, 'rb') as f:
>                 part = 0
>                 while True:
>                     chunk = f.read(MAX_SIZE)
>                     if not chunk:
>                         break
>                     if count >= MAX_FILES:
>                         new_bucket()
>                     chunkname = '%s.part%03d' % (name, part)
>                     dst = os.path.join(RESHARD, '%03d' % bucket_idx, chunkname)
>                     with open(dst, 'wb') as out:
>                         out.write(chunk)
>                     parts.append('%03d/%s' % (bucket_idx, chunkname))
>                     count += 1
>                     part += 1
>             os.remove(src)
>         manifest.append({'orig': name, 'parts': parts})
>
>     with open(os.path.join(RESHARD, 'manifest.json'), 'w') as f:
>         json.dump(manifest, f)
>
>     print('done: %d files -> %d buckets' % (len(files), bucket_idx + 1))
>
>
> if __name__ == '__main__':
>     main()
> PYEOF
