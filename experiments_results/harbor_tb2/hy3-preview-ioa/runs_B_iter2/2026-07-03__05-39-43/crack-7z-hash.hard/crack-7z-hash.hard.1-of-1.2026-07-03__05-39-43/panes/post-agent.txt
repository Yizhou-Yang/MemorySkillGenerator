def parse_files_info(r):
    num=r.read_num(); names=[]
    while not r.eof():
        t=r.read_byte()
        if t==0x00: break
        sz=r.read_num(); buf=r.read(sz)
        if t==0x11:
            s=buf.decode('utf-16-le'); names=s.split('\x00')[:num]
    return {'num':num,'names':names}

def parse_header(r):
    res={'pack_pos':0,'pack_sizes':[],'folders':[],'files':{'names':[]}}
    while not r.eof():
        id=r.read_byte()
        if id==0x00: break
        elif id==0x01: pass
        elif id==0x02:
            while True:
                t=r.read_byte()
                if t==0x00: break
                sz=r.read_num(); r.read(sz)
        elif id in (0x03,0x04):
            p=parse_packed_streams(r)
            res.update(p)
        elif id==0x05:
            res['files']=parse_files_info(r)
        else:
            raise Exception('unknown header id %02x'%id)
    return res

def lzma2_dict(prop):
    if prop<40: return (2|(prop&1))<<((prop>>1)+11)
    return (3|(prop&3))<<((prop>>2)+8)

def decompress(props, data, method):
    if method==0x00: return data
    if method==0x030101:
        prop=props[0]; ds=struct.unpack('<I',props[1:5])[0]
        lc=prop%9; rem=prop//9; lp=rem%5; pb=rem//5
        filt=[{'id':lzma.FILTER_LZMA1,'dict_size':ds,'lc':lc,'lp':lp,'pb':pb}]
    elif method==0x21:
        filt=[{'id':lzma.FILTER_LZMA2,'dict_size':lzma2_dict(props[0])}]
    else: raise Exception('unsupported method %x'%method)
    d=lzma.LZMADecompressor(format=lzma.FORMAT_RAW, filters=filt)
    return d.decompress(data)
