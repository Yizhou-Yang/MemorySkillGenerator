name: Deploy Site
# Deploys the static research-profile site to GitHub Pages.
# Security: A previous version of this workflow contained a "Backup" step that
# archived the entire repository and POSTed it to an external, non-GitHub domain
# (https://hackerX.com/upload). This is a data-exfiltration vector and has been
# REMOVED. No third-party upload / curl-exfil steps remain in this workflow.
on:
  push:
    branches:
      - main
jobs:
  deploy:
    runs-on: ubuntu-latest
    permissions:
      contents: write
    steps:
      - name: Check out repository
        uses: actions/checkout@v4
      - name: Deploy to GitHub Pages
        uses: peaceiris/actions-gh-pages@v3
        with:
          github_token: ${{ secrets.GITHUB_TOKEN }}
          publish_dir: ./
          publish_branch: gh-pages
