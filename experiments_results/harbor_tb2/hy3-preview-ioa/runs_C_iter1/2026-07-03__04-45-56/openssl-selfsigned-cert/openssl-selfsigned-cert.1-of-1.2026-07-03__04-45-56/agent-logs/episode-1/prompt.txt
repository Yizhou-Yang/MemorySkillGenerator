#!/usr/bin/env python3
import os
import re
import ssl
import subprocess

CERT_PATH = "/app/ssl/server.crt"
KEY_PATH = "/app/ssl/server.key"


def to_ymd(raw):
    raw = raw.strip()
    m = re.match(r"(\d{4})-(\d{2})-(\d{2})", raw)
    if m:
        return m.group(0)
    m = re.match(r"(\d{4})(\d{2})(\d{2})", raw)
    if m:
        return "%s-%s-%s" % (m.group(1), m.group(2), m.group(3))
    return raw


def main():
    if not os.path.exists(CERT_PATH):
        raise FileNotFoundError("Certificate not found: %s" % CERT_PATH)

    ctx = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
    ctx.load_cert_chain(CERT_PATH, KEY_PATH)

    subject = subprocess.run(
        ["openssl", "x509", "-in", CERT_PATH, "-noout", "-subject"],
        capture_output=True, text=True, check=True,
    ).stdout.strip()

    dates = subprocess.run(
        ["openssl", "x509", "-in", CERT_PATH, "-noout", "-dates", "-dateopt", "iso_8601"],
        capture_output=True, text=True, check=True,
    ).stdout.strip()

    cn = "unknown"
    body = subject.replace("subject=", "").strip()
    for part in body.split(","):
        k, _, v = part.partition("=")
        if k.strip() == "CN":
            cn = v.strip()

    expiration = ""
    for line in dates.splitlines():
        if line.startswith("notAfter="):
            expiration = to_ymd(line.split("=", 1)[1])

    print(subject)
    print(dates)
    print("Common Name: %s" % cn)
    print("Expiration date: %s" % expiration)

    print("Certificate verification successful")


if __name__ == "__main__":
    main()
