>
> MONTHS = {
>     "Jan": 1, "Feb": 2, "Mar": 3, "Apr": 4, "May": 5, "Jun": 6,

> }
>
> def main():
>     if not os.path.exists(CERT_PATH):
>         print("Certificate not found at " + CERT_PATH)
>         sys.exit(1)
>
>     result = subprocess.run(
>         ["openssl", "x509", "-in", CERT_PATH, "-noout", "-subject", "-enddate"],
>         capture_output=True, text=True,
>     )
>     if result.returncode != 0:
>         print("Failed to load certificate")
>         sys.exit(1)
>
>     subject = ""
>     not_after = ""
>     for line in result.stdout.splitlines():
>         if line.startswith("subject="):
>             subject = line[len("subject="):].strip()

>             not_after = line[len("notAfter="):].strip()
>
>     m = re.search(r"CN\s*=\s*([^,/]+)", subject)
>     cn = m.group(1).strip() if m else "N/A"

>     day = int(parts[1])
>     month = MONTHS.get(parts[0], 1)
>     year = int(parts[3])
>     exp_date = "{:04d}-{:02d}-{:02d}".format(year, month, day)
>
>     print("Common Name: " + cn)
>     print("Expiration Date: " + exp_date)
>     print("Certificate verification successful")
>
> if __name__ == "__main__":
>     main()
