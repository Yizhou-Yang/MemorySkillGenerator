> import os
> import torch
> import torch.nn.functional as F
> from flask import Flask, request, jsonify
> from transformers import AutoTokenizer, AutoModelForSequenceClassification
>
> MODEL_PATH = "/app/model_cache/sentiment_model"
>
> app = Flask(__name__)
>
> print("Loading model from", MODEL_PATH, flush=True)
> tokenizer = AutoTokenizer.from_pretrained(MODEL_PATH, local_files_only=True)
> model = AutoModelForSequenceClassification.from_pretrained(MODEL_PATH, local_files_only=True)
> model.eval()
> print("Model loaded.", flush=True)
>
> # distilbert sst-2 config: id2label -> 0=NEGATIVE, 1=POSITIVE
> id2label = getattr(model.config, "id2label", None)
> if id2label and isinstance(id2label, dict):
>     label_map = {int(k): str(v).lower() for k, v in id2label.items()}
> else:
>     label_map = {0: "negative", 1: "positive"}
>
>
> def analyze(text):
>     inputs = tokenizer(text, return_tensors="pt", truncation=True, max_length=512)
>     with torch.no_grad():
>         logits = model(**inputs).logits
>         probs = F.softmax(logits, dim=-1)[0]
>     conf = {}
>     for idx, label in label_map.items():
>         # normalize label key to positive/negative
>         key = "positive" if "pos" in label else ("negative" if "neg" in label else label)
>         conf[key] = float(probs[idx].item())
>     if "positive" not in conf:
>         conf["positive"] = 0.0
>     if "negative" not in conf:
>         conf["negative"] = 0.0
>     sentiment = "positive" if conf["positive"] >= conf["negative"] else "negative"
>     return sentiment, conf
>
>
> @app.route("/sentiment", methods=["POST"])
> def sentiment():
>     data = request.get_json(silent=True)
>     if data is None:
>         return jsonify({"error": "Request body must be valid JSON"}), 400
>     if not isinstance(data, dict) or "text" not in data:
>         return jsonify({"error": "Missing required field 'text'"}), 400
>     text = data["text"]
>     if not isinstance(text, str) or not text.strip():
>         return jsonify({"error": "Field 'text' must be a non-empty string"}), 400
>     try:
>         sentiment_label, conf = analyze(text)
>         return jsonify({"sentiment": sentiment_label, "confidence": conf})
>     except Exception as e:
>         return jsonify({"error": str(e)}), 500
>
>
> @app.route("/health", methods=["GET"])
> def health():
>     return jsonify({"status": "ok"})
>
>
> if __name__ == "__main__":
>     app.run(host="0.0.0.0", port=5000)
> EOF
