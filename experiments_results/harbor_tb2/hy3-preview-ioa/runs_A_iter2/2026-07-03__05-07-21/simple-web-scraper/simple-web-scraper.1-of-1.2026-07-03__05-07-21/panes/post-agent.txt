> import re
> import requests
> from bs4 import BeautifulSoup
> import pandas as pd
>
> URL = "http://server:8000"
>
>
> def parse_num(text, pattern):
>     m = re.search(pattern, text)
>     return m.group(1) if m else ""
>
>
> def scrape(url=URL):
>     resp = requests.get(url, timeout=30)
>     resp.raise_for_status()
>     soup = BeautifulSoup(resp.text, "html.parser")
>
>     books = []
>     for book in soup.find_all("div", class_="book"):
>         title_el = book.find("div", class_="book-title")
>         author_el = book.find("div", class_="book-author")
>         year_el = book.find("span", class_="book-year")
>         rating_el = book.find("span", class_="book-rating")
>         price_el = book.find("span", class_="book-price")
>
>         title = title_el.get_text(strip=True) if title_el else ""
>         author = author_el.get_text(strip=True) if author_el else ""
>         # remove leading 'by '
>         author = re.sub(r"^\s*by\s+", "", author)
>         year = parse_num(year_el.get_text(strip=True), r"(\d{4})") if year_el else ""
>         rating = parse_num(rating_el.get_text(strip=True), r"([\d.]+)/?5?") if rating_el else ""
>         price = parse_num(price_el.get_text(strip=True), r"\$?([\d.]+)") if price_el else ""
>
>         books.append({
>             "title": title,
>             "author": author,
>             "year": int(year) if year else "",
>             "rating": float(rating) if rating else "",
>             "price": float(price) if price else "",
>         })
>     return books
>
>
> def main():
>     books = scrape()
>     df = pd.DataFrame(books, columns=["title", "author", "year", "rating", "price"])
>     df.to_csv("books.csv", index=False)
>
>     total = len(df)
>     avg_price = df["price"].mean()
>     idx = df["rating"].idxmax()
>     top = df.loc[idx]
>
>     report = (
>         "Book Store Scraping Summary Report\n"
>         "==================================\n"
>         f"Total number of books scraped: {total}\n"
>         f"Average book price: ${avg_price:.2f}\n"
>         f"Highest rated book: {top['title']} by {top['author']} "
>         f"(Rating: {top['rating']}/5, Price: ${top['price']:.2f})\n"
>     )
>     with open("report.txt", "w") as f:
>         f.write(report)
>
>     print(report)
>
>
> if __name__ == "__main__":
>     main()
> EOF
