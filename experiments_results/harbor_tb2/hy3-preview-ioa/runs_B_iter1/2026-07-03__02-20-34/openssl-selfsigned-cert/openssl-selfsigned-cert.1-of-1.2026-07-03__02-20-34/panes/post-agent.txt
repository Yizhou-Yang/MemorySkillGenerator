> import os
> import ssl
> import subprocess
>
> CERT_PATH = "/app/ssl/server.crt"
>
>
> def main():
>     if not os.path.exists(CERT_PATH):
>         print("Certificate not found at " + CERT_PATH)
>         return
>
>     result = subprocess.run(
>         ["openssl", "x509", "-in", CERT_PATH, "-noout", "-subject", "-dates"],
>         capture_output=True, text=True,
>     )
>     if result.returncode != 0:
>         print("Failed to load certificate")
>         print(result.stderr)
>         return
>
>     output = result.stdout
>
>     # Extract Common Name
>     common_name = "unknown"
>     for line in output.splitlines():
>         if line.strip().startswith("subject="):
>             subj = line.split("subject=", 1)[1]
>             for part in subj.split(","):
>                 part = part.strip()
>                 if part.startswith("CN ="):
>                     common_name = part.split("CN =", 1)[1].strip()
>
>     # Extract notAfter and convert to YYYY-MM-DD
>     expiration = "unknown"
>     for line in output.splitlines():
>         if line.strip().startswith("notAfter="):
>             date_str = line.split("notAfter=", 1)[1].strip()  # e.g. Jul  2 19:40:36 2027 GMT
>             try:
>                 from datetime import datetime
>                 dt = datetime.strptime(date_str, "%b %d %H:%M:%S %Y %Z")
>                 expiration = dt.strftime("%Y-%m-%d")
>             except Exception:
>                 expiration = date_str
>
>     print("Common Name: " + common_name)
>     print("Expiration Date: " + expiration)
>     print("Certificate verification successful")
>
>
> if __name__ == "__main__":
>     main()
> EOF
