> from PIL import Image
> import numpy as np
> im = Image.open('/app/chess_puzzle.png').convert('RGB')
> a = np.array(im).astype(int)
> cell=80
> files='abcdefgh'
> cells=[]
> masks=[]
> for r in range(8):
>     for c in range(8):
>         sub=a[r*cell:(r+1)*cell, c*cell:(c+1)*cell]
>         bg=sub[2,2].astype(int)
>         d=np.abs(sub.astype(int)-bg).sum(axis=2)
>         mask=(d>40)
>         if mask.mean()>0.03:
>             ys,xs=np.where(mask)
>             y0,y1,x0,x1=ys.min(),ys.max()+1,xs.min(),xs.max()+1
>             m=mask[y0:y1,x0:x1]
>             m=Image.fromarray((m*255).astype('uint8')).resize((32,32))
>             m=np.array(m)>80
>             cells.append((files[c],8-r))
>             masks.append(m)
> # classify color
> colors=[]
> for r in range(8):
>     for c in range(8):
>         pass
> # recompute color by RGB
> for (fc,rr) in cells:
>     c=files.index(fc); r=8-rr
>     sub=a[r*cell:(r+1)*cell, c*cell:(c+1)*cell]
>     bg=sub[2,2].astype(int)
>     d=np.abs(sub.astype(int)-bg).sum(axis=2)
>     cols=sub[d>40]
>     lum=cols.mean()
>     colors.append('W' if lum>140 else 'B')
> # cluster
> N=len(cells)
> dist=[[0]*N for _ in range(N)]
> for i in range(N):
>     for j in range(N):
>         dist[i][j]=(masks[i]^masks[j]).mean()
> assigned=[-1]*N; reps=[]
> for i in range(N):
>     if assigned[i]!=-1: continue
>     assigned[i]=len(reps)
>     for j in range(i+1,N):
>         if assigned[j]==-1 and dist[i][j]<0.10:
>             assigned[j]=len(reps)
>     reps.append([i])
> print('num clusters', len(reps))
> for k,grp in enumerate(reps):
>     print('--- cluster',k,'members:',[ (cells[i],colors[i]) for i in grp])
>     rep=masks[grp[0]]
>     for y in range(32):
>         print(''.join('#' if rep[y,x] else ' ' for x in range(32)))
> EOF
