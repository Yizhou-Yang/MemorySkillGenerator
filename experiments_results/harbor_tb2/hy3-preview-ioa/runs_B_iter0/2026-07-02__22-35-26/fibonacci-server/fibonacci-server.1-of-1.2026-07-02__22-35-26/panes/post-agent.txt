> import json
> from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
> from urllib.parse import urlparse, parse_qs

> def fib(n):
>     if n < 0:
>         raise ValueError("n must be non-negative")
>     a, b = 0, 1
>     for _ in range(n):
>         a, b = b, a + b
>     return a
>
>
> class Handler(BaseHTTPRequestHandler):
>     def do_GET(self):
>         parsed = urlparse(self.path)
>         if parsed.path != "/fib":
>             self.send_response(404)
>             self.send_header("Content-Type", "application/json")
>             self.end_headers()
>             self.wfile.write(json.dumps({"error": "not found"}).encode())
>             return
>         qs = parse_qs(parsed.query)
>         if "n" not in qs:
>             self.send_error(400, "Missing query parameter 'n'")
>             return
>         raw = qs["n"][0]
>         try:
>             n = int(raw)
>         except (ValueError, TypeError):
>             self.send_error(400, "Query parameter 'n' must be an integer")
>             return
>         try:
>             result = fib(n)
>         except ValueError:
>             self.send_error(400, "Query parameter 'n' must be a non-negative integer")
>             return
>         body = json.dumps({"result": result}).encode()
>         self.send_response(200)
>         self.send_header("Content-Type", "application/json")
>         self.send_header("Content-Length", str(len(body)))
>         self.end_headers()
>         self.wfile.write(body)
>
>     def log_message(self, *args):
>         pass
>
>
> def main():
>     server = ThreadingHTTPServer(("0.0.0.0", 3000), Handler)
>     server.serve_forever()
>
>

>     main()
