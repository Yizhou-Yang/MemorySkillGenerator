    with urllib.request.urlopen(req, timeout=180) as r, open(fn,'wb') as f:

            chunk = r.read(1<<20)

    sol = d['deepseek_solution']
    n = len(domain)
    texts = []
    for i in range(n):
        if (domain[i] or '').strip() == 'science':
            texts.append((reason[i] or '') + (sol[i] or ''))
    science_rows += len(texts)
    for j in range(0, len(texts), 500):
        encs = tok.encode_batch(texts[j:j+500])
        for e in encs:
            total += len(e.ids)
    del texts
    os.remove(fn)
    print('running total', total, 'science_rows', science_rows, flush=True)
print('FINAL', total)
