#!/usr/bin/env python3
import os, sys

base = os.path.dirname(os.path.abspath(__file__))
reshard = os.path.join(base, "c4_reshard")
sample = os.path.join(base, "c4_sample")

def main():
    chunks = {}
    for root, dirs, files in os.walk(reshard):
        for fn in files:
            base_name, dot, idx = fn.rpartition('.')
            if not dot:
                continue
            if not idx.isdigit():
                continue
            chunks.setdefault(base_name, []).append((int(idx), os.path.join(root, fn)))
    for base_name, lst in chunks.items():
        lst.sort(key=lambda x: x[0])
        out_path = os.path.join(sample, base_name)
        os.makedirs(sample, exist_ok=True)
        with open(out_path, 'wb') as out:
            for _, p in lst:
                with open(p, 'rb') as inp:
                    while True:
                        buf = inp.read(1024*1024)
                        if not buf: break
                        out.write(buf)
        print("reconstructed", out_path)

if __name__ == "__main__":
    main()
