import os
import shutil

ROOT = "/app"
SAMPLE = os.path.join(ROOT, "c4_sample")
RESHARD = os.path.join(ROOT, "c4_reshard")

PER_LEAF = 30    # max files per leaf folder
PER_GROUP = 30   # max leaf folders per group folder

files = sorted(
    f for f in os.listdir(SAMPLE)
    if os.path.isfile(os.path.join(SAMPLE, f))
)
n = len(files)
leaves = (n + PER_LEAF - 1) // PER_LEAF
groups = (leaves + PER_GROUP - 1) // PER_GROUP
print("files:", n, "leaves:", leaves, "groups:", groups)

if os.path.exists(RESHARD):
    shutil.rmtree(RESHARD)
os.makedirs(RESHARD)

for i, fname in enumerate(files):
    leaf = i // PER_LEAF
    group = leaf // PER_GROUP
    leaf_in_group = leaf % PER_GROUP
    dest_dir = os.path.join(RESHARD, "%03d" % group, "%03d" % leaf_in_group)
    os.makedirs(dest_dir, exist_ok=True)
    os.link(os.path.join(SAMPLE, fname), os.path.join(dest_dir, fname))

print("done")
PYEOF
