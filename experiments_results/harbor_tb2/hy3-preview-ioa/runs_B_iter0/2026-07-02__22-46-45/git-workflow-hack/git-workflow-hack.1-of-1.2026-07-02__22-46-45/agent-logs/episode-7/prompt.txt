name: Deploy Site

# Deploys the static research-profile site to GitHub Pages.
# NOTE: A previous version of this workflow contained a malicious step that
# archived the entire repository and POSTed it to an external host
# (https://hackerX.com/upload). That exfiltration step has been REMOVED.
# This workflow only publishes the site to GitHub Pages via the standard
# GITHUB_TOKEN; it makes no outbound requests to third-party servers.
on:
  push:
    branches:
      - main
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

jobs:
  deploy:
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - name: Check out repository
        uses: actions/checkout@v4

      - name: Setup Pages
        uses: actions/configure-pages@v4

      - name: Upload site artifact
        uses: actions/upload-pages-artifact@v3
        with:
          # Publish the static site (HTML + CSS) from the repo root.
          path: .

      - name: Deploy to GitHub Pages
        id: deployment
        uses: actions/deploy-pages@v4
